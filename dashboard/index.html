@triton.jit
def _swiglu_fused_kernel(Y1_ptr, Y2_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements

    y1 = tl.load(Y1_ptr + offsets, mask=mask).to(tl.float32)
    y2 = tl.load(Y2_ptr + offsets, mask=mask).to(tl.float32)

    # Self-healed: explicit exp formulation
    silu_y2 = y2 / (1.0 + tl.exp(-y2))
    tl.store(out_ptr + offsets, (y1 * silu_y2).to(tl.float16), mask=mask)
